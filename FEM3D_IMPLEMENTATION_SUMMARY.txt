================================================================================
4. MATERIAL MODEL EQUATIONS
================================================================================

LINEAR ELASTIC:
  D matrix (isotropic):
    c1 = E/((1+nu)(1-2*nu)) * (1-nu)
    c2 = c1 * nu
    c3 = c1 * (1-2*nu) / 2
    D[0,0]=D[1,1]=D[2,2]=c1, D[0,1]=D[1,0]=...=c2, D[3,3]=D[4,4]=D[5,5]=c3

NEO-HOOKEAN:
  Lame parameters:
    mu = E / (2(1+nu))
    lambda = E*nu / ((1+nu)(1-2*nu))
  
  Deformation:
    F = current_grad / reference_grad (3x3 tensor)
    J = det(F)
    B = F * F^T (left Cauchy-Green)
  
  Cauchy stress:
    sigma = (mu/J)(B - I) + (lambda/J)*ln(J)*I
  
  Material tangent (isotropic):
    K = lambda + 2*mu/3
    G = mu
    c1 = K - 2*G/3
    c2 = 2*G

================================================================================
5. CONTACT MECHANICS FORMULATION
================================================================================

Gap Function:
  g = ||x_slave - x_master||
  Contact occurs when: g < contact_tolerance

Penalty Force:
  f_contact = -penalty_stiffness * max(0, tolerance - gap) * normal

Normal Direction:
  n = (x_master - x_slave) / ||x_master - x_slave||

Contact Stiffness:
  K_contact[dof,dof] += penalty_stiffness (diagonal)

Nearest Point Search:
  - Dense search for node-to-node (O(n*m))
  - KDTree search for node-to-surface (O(n*log(m)))

7. EXAMPLE USAGE
================================================================================

from fem_3d_solver import (FEM3DSolver, MaterialRegion, BoundaryCondition, 
                           ContactPair, FEM3DResults)
import numpy as np

# Create solver (nonlinear by default)
solver = FEM3DSolver(verbose=True, nonlinear=True)

# Define mesh
nodes = np.array([[0,0,0], [1,0,0], [0,1,0], [0,0,1], ...])  # (n_nodes, 3)
elements = np.array([[0,1,2,3], [1,2,3,4], ...])  # (n_elements, 4)

# Define materials
bone = MaterialRegion(
    name='bone', E=10000, nu=0.3, 
    element_ids=np.array([0,1,2,...]),
    model='linear'
)
cartilage = MaterialRegion(
    name='cartilage', E=5, nu=0.4,
    element_ids=np.array([...]),
    model='neo_hookean'
)

# Define boundary conditions
bc_fixed = BoundaryCondition(
    node_ids=np.array([10, 11, 12]),
    dof_mask=np.array([True, True, True]),
    prescribed_disp=None
)

# Define contact
contact = ContactPair(
    master_surface_nodes=np.array([...]),
    slave_surface_nodes=np.array([...]),
    penalty_stiffness=500.0,
    contact_tolerance=2.0
)

# Run analysis
results = solver.analyze(
    nodes=nodes,
    elements=elements,
    material_regions=[bone, cartilage],
    boundary_conditions=[bc_fixed],
    contact_pairs=[contact]
)

# Access results
print(f"Peak von Mises: {results.von_mises_stress.max():.2f} MPa")
print(f"Peak contact pressure: {results.peak_contact_pressure:.2f} MPa")
print(f"Converged: {results.converged} in {results.n_iterations} iterations")
